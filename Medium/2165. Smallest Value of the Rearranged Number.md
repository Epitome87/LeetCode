# [2165. Smallest Value of the Rearranged Number](https://leetcode.com/problems/smallest-value-of-the-rearranged-number)

## Description

You are given an integer `num.` **Rearrange** the digits of `num` such that its value is **minimized** and it does not contain **any** leading zeros.

Return _the rearranged number with minimal value_.

Note that the sign of the number does not change after rearranging the digits.

**Example 1:**

**Input:** num = 310  
**Output:** 103  
**Explanation:** The possible arrangements for the digits of 310 are 013, 031, 103, 130, 301, 310.  
The arrangement with the smallest value that does not contain any leading zeros is 103.

**Example 2:**

**Input:** num = -7605  
**Output:** -7650  
**Explanation:** Some possible arrangements for the digits of -7605 are -7650, -6705, -5076, -0567.  
The arrangement with the smallest value that does not contain any leading zeros is -7650.

**Constraints:**

- `-10^15 <= num <= 10^15`

## My Solution

**JavaScript**

```js
// Approach: Counting Sort
const smallestNumber = (num) => {
  if (num === 0) return 0;

  const isNegative = num < 0;
  num = Math.abs(num);

  // Frequency for digits 0-9
  const counts = new Array(10).fill(0);
  while (num > 0) {
    counts[num % 10]++;
    num = Math.floor(num / 10);
  }

  let result = 0;

  if (isNegative) {
    // Build the largest possible number (descending)
    for (let i = 9; i >= 0; --i) {
      while (counts[i]-- > 0) {
        result = result * 10 + i;
      }
    }
    return -result;
  }

  // Otherwise, number is positive, so build the smallest possible number (ascending)
  // First, we need to find the first non-zero digit to place at the front
  for (let i = 1; i <= 9; ++i) {
    if (counts[i] > 0) {
      result = i;
      counts[i]--;
      break;
    }
  }

  // With the (potential) 0 in the front, build the rest of the number ascending
  for (let i = 0; i <= 9; ++i) {
    while (counts[i]-- > 0) {
      result = result * 10 + i;
    }
  }

  return result;
};
```

```js
const smallestNumber = (num) => {
  const isNegative = num < 0;
  const digits = [...`${Math.abs(num)}`].sort((a, b) => (isNegative ? b - a : a - b));

  if (!isNegative && digits[0] === '0') {
    const firstNonZero = digits.findIndex((d) => d !== '0');
    if (firstNonZero !== -1) {
      [digits[0], digits[firstNonZero]] = [digits[firstNonZero], digits[0]];
    }
  }

  const result = Number(digits.join(''));
  return isNegative ? -result : result;
};
```

```js
// Approach: For positive numbers, greedily move the lowest digit to the front of the number.
// Ensure we do not move a 0 to the very front, though.
// For negative numbers, greedily move the highest digit to the front of the number.
const smallestNumber = (num) => {
  if (num === 0) return 0;

  const isNegative = num < 0;
  num = Math.abs(num);
  const digits = [];

  // Collect the digits without using built-in methods
  while (num > 0) {
    digits.push(num % 10);
    num = Math.floor(num / 10);
  }

  if (isNegative) digits.sort((a, b) => b - a);
  else {
    digits.sort((a, b) => a - b);

    // Ensure we do not place a 0 in the front for positive numbers
    if (digits[0] === 0) {
      // Find the index of the first non-0
      let firstNonZero = 0;
      while (firstNonZero < digits.length && digits[firstNonZero] === 0) {
        ++firstNonZero;
      }

      // If a non-0 was found, swap it to the front
      if (firstNonZero < digits.length) {
        [digits[0], digits[firstNonZero]] = [digits[firstNonZero], digits[0]];
      }
    }
  }

  let result = 0;
  for (const digit of digits) {
    result = result * 10 + digit;
  }

  return isNegative ? -result : result;
};
```
